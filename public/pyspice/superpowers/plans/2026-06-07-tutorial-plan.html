# 初升大教程

欢迎来到 PySpice 初升大教程！本教程专为**初中、高中和高校学生**设计，从零开始教学如何使用 Python 和 PySpice 进行电路仿真。

## 这个教程适合谁？

- 🎒 **初中生**：刚开始学习物理电学，想用电脑模拟课堂实验
- 📚 **高中生**：正在学习电磁学，希望动手验证复杂的电路定律
- 🎓 **高校学生**：需要深入理解电路分析、模拟电子和数字逻辑

## 你需要准备什么？

- 一台能运行 Python 的电脑（Windows、macOS 或 Linux）
- 对电路和物理的基本好奇心
- （可选）会一点 Python 更好，但不会也没关系——我们会一步步教你

## 教程结构

本教程按照中国**人教版物理教材**的知识顺序编排，分为三个级别：

| 级别 | 章节数 | 适合人群 |
|------|--------|----------|
| 📗 [初中篇](/tutorial/middle-school/) | 6 章 | 八年级、九年级学生 |
| 📘 [高中篇](/tutorial/high-school/) | 11 章 | 高一至高三学生 |
| 📙 [高校篇](/tutorial/college/) | 17 章 | 大学理工科学生 |

## 如何使用本教程？

每一章都遵循相同的结构：

1. **本节你将学到** — 明确的学习目标
2. **知识回顾/情景引入** — 从日常生活出发，建立物理直觉
3. **动手设计电路** — 先画出电路图，手工计算预期结果
4. **用 Python 实现电路仿真** — 用代码构建电路、运行仿真、分析结果
5. **深入理解** — 对比手工计算与仿真结果
6. **本章小结** — 核心要点回顾
7. **练习** — 动手巩固

> 💡 **提示**：如果在某个知识点上卡住了，不要着急。试着回头复习一下前面的章节，或者在[电路课本](https://baike.baidu.com/)中查找相关概念。

Python script (ch01_circuit_current.py):

python
import PySpice.Logging.Logging as Logging
logger = Logging.setup_logging()

from PySpice.Spice.Netlist import Circuit
from PySpice.Unit import *

# 第1章：电路与电流
# 构建一个最简单的电路：电池 + 灯泡（电阻）+ 开关（导线）

circuit = Circuit('Simple Circuit')

# 10V 电池，正极连节点 1，负极连地（节点 0）
circuit.V('input', 1, circuit.gnd, 10@u_V)

# 灯泡可以看作一个电阻，我们用一个 100Ω 的电阻表示
circuit.R('lamp', 1, circuit.gnd, 100@u_Ω)

# 使用仿真器分析电路的工作点（直流工作状态）
simulator = circuit.simulator(temperature=25, nominal_temperature=25,
                              simulator='ngspice-subprocess')
analysis = simulator.operating_point()

# 读取结果
print("=" * 50)
print("电路仿真结果：简单电路")
print("=" * 50)
for node in analysis.nodes.values():
    voltage = float(node[0])
    print("节点 {} 的电压: {:6.2f} V".format(str(node), voltage))

# 计算电流
voltage = float(analysis.nodes['1'][0])
resistance = 100
current = voltage / resistance
print("\n根据欧姆定律 I = V / R:")
print("电流 I = {:.2f} V / {} Ω = {:.3f} A".format(voltage, resistance, current))
print("电流 I = {:.2f} mA".format(current * 1000))

Python script (ch02_voltage_resistance.py):

python
import PySpice.Logging.Logging as Logging
logger = Logging.setup_logging()

from PySpice.Spice.Netlist import Circuit
from PySpice.Unit import *

# 第2章：电压与电阻
# 探究电压、电阻和电流之间的关系
# 使用多个电阻串联，验证电阻的概念

# 电路1：单个电阻
circuit1 = Circuit('Single Resistor')
circuit1.V('input', 1, circuit1.gnd, 12@u_V)
circuit1.R(1, 1, circuit1.gnd, 100@u_Ω)

sim1 = circuit1.simulator(temperature=25, nominal_temperature=25,
                          simulator='ngspice-subprocess')
analysis1 = sim1.operating_point()

print("=" * 50)
print("电路1：单个 100Ω 电阻，12V 电源")
print("=" * 50)
v = float(analysis1.nodes['1'][0])
print("电阻两端电压: {:.2f} V".format(v))
print("流过电阻的电流: {:.3f} A = {:.1f} mA".format(v/100, v/100*1000))

# 电路2：两个电阻串联
circuit2 = Circuit('Two Resistors in Series')
circuit2.V('input', 1, circuit2.gnd, 12@u_V)
circuit2.R(1, 1, 2, 200@u_Ω)
circuit2.R(2, 2, circuit2.gnd, 100@u_Ω)

sim2 = circuit2.simulator(temperature=25, nominal_temperature=25,
                          simulator='ngspice-subprocess')
analysis2 = sim2.operating_point()

print("\n" + "=" * 50)
print("电路2：两个电阻串联 (200Ω + 100Ω)，12V 电源")
print("=" * 50)
v1 = float(analysis2.nodes['1'][0])
v2 = float(analysis2.nodes['2'][0])
print("节点 1 (电源端) 电压: {:.2f} V".format(v1))

python
import PySpice.Logging.Logging as Logging

from PySpice.Spice.Netlist import Circuit
from PySpice.Unit import *
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# 第3章：欧姆定律
# 通过扫描电压来绘制电阻的 I-V 特性曲线

circuit = Circuit('Ohms Law Demo')
circuit.V('input', 1, circuit.gnd, 1@u_V)
circuit.R(1, 1, circuit.gnd, 100@u_Ω)

                              simulator='ngspice-subprocess')

# 提取数据
voltage = analysis['1']

print("=" * 50)
print("欧姆定律验证：I-V 特性曲线")
print("=" * 50)

print("电阻值: 100 Ω")

print("-" * 45)

    c = float(current[i])
    print("{:6.2f}    {:10.6f}   {:10.6f}".format(v, c, v/100))

plt.figure(figsize=(8, 5))
plt.plot(voltage, current * 1000, 'b-', linewidth=2, label='仿真结果')
plt.plot(voltage, voltage / 100 * 1000, 'r--', linewidth=1, label='理论值 I=V/100Ω')
plt.xlabel('电压 (V)')
plt.ylabel('电流 (mA)')
plt.title('欧姆定律：电阻 I-V 特性曲线')

import PySpice.Logging.Logging as Logging
logger = Logging.setup_logging()

from PySpice.Spice.Netlist import Circuit
from PySpice.Unit import *

# 第4章：串并联电路
# 对比串联和并联电路中电流、电压的分布

vsupply = 12@u_V
r1_val = 100@u_Ω
r2_val = 200@u_Ω

# ---- 串联电路 ----
circuit_series = Circuit('Series Circuit')
circuit_series.V('input', 1, circuit_series.gnd, vsupply)
circuit_series.R(1, 1, 2, r1_val)
circuit_series.R(2, 2, circuit_series.gnd, r2_val)

sim_series = circuit_series.simulator(temperature=25, nominal_temperature=25,
                                      simulator='ngspice-subprocess')
analysis_series = sim_series.operating_point()

print("=" * 50)
print("串联电路分析 (12V, R1=100Ω, R2=200Ω)")
print("=" * 50)
v1 = float(analysis_series.nodes['1'][0])
v2 = float(analysis_series.nodes['2'][0])
print("节点 1 电压: {:.2f} V".format(v1))
print("节点 2 电压: {:.2f} V".format(v2))
print("R1 两端电压: {:.2f} V".format(v1 - v2))
print("R2 两端电压: {:.2f} V".format(v2))
i_series = (v1 - v2) / 100
print("串联电流: {:.3f} A = {:.1f} mA".format(abs(i_series), abs(i_series)*1000))
print("等效电阻 (串联): {} Ω + {} Ω = {} Ω".format(100, 200, 100+200))

# ---- 并联电路 ----
circuit_parallel = Circuit('Parallel Circuit')
circuit_parallel.V('input', 1, circuit_parallel.gnd, vsupply)
circuit_parallel.R(1, 1, circuit_parallel.gnd, r1_val)
circuit_parallel.R(2, 1, circuit_parallel.gnd, r2_val)

sim_parallel = circuit_parallel.simulator(temperature=25, nominal_temperature=25,
                                          simulator='ngspice-subprocess')
analysis_parallel = sim_parallel.operating_point()

print("\n" + "=" * 50)
print("并联电路分析 (12V, R1=100Ω, R2=200Ω)")
print("=" * 50)
v = float(analysis_parallel.nodes['1'][0])
print("节点 1 电压: {:.2f} V".format(v))
i1 = v / 100
i2 = v / 200
print("流过 R1 (100Ω) 的电流: {:.3f} A = {:.1f} mA".format(i1, i1*1000))
print("流过 R2 (200Ω) 的电流: {:.3f} A = {:.1f} mA".format(i2, i2*1000))
print("总电流: {:.3f} A = {:.1f} mA".format(i1+i2, (i1+i2)*1000))
req = 1 / (1/100 + 1/200)
print("等效电阻 (并联): 1/(1/100 + 1/200) = {:.2f} Ω".format(req))

# ---- 对比总结 ----
print("\n" + "=" * 50)
print("串并联对比总结")
print("=" * 50)
print("串联特点：电流相同，电压按电阻比例分配")
print("并联特点：电压相同，电流按电阻反比分配")
print("串联等效电阻增加，并联等效电阻减小")

import PySpice.Logging.Logging as Logging
logger = Logging.setup_logging()

from PySpice.Spice.Netlist import Circuit
from PySpice.Unit import *
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# 第5章：电功与电功率
# P = UI = I²R = U²/R
# 通过 DC 扫描观察功率随电压的变化

ax2.set_xlabel('电压 (V)')
ax2.set_ylabel('功率 (mW)')
ax2.set_title('P = U²/R — 功率随电压平方增长')
ax2.grid(True, alpha=0.3)

Python script (ch06_joule_safety.py):

python
import PySpice.Logging.Logging as Logging
logger = Logging.setup_logging()

from PySpice.Spice.Netlist import Circuit
from PySpice.Unit import *
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt

# 第6章：焦耳定律与安全用电
# Q = I²Rt，演示电流热效应和短路危害

# 正常电路
circuit_normal = Circuit('Normal Circuit')
circuit_normal.V('input', 1, circuit_normal.gnd, 220@u_V)
circuit_normal.R('load', 1, circuit_normal.gnd, 484@u_Ω)

sim = circuit_normal.simulator(temperature=25, nominal_temperature=25,
                               simulator='ngspice-subprocess')
analysis = sim.operating_point()

v = float(analysis.nodes['1'][0])
i = v / 484
p = v * i

print("=" * 50)
print("焦耳定律与安全用电")
print("=" * 50)
print()
print("【正常电路】家用电器 (220V, 484Ω)")
print("  电流 I = {:.2f} A".format(i))
print("  功率 P = {:.2f} W".format(p))
print("  每秒发热 Q = {:.2f} J".format(p))  # P*t, t=1s
print("  (这相当于一个 100W 的灯泡)")

# 短路电路：电阻极小
circuit_short = Circuit('Short Circuit')
circuit_short.V('input', 1, circuit_short.gnd, 220@u_V)
circuit_short.R('wire', 1, circuit_short.gnd, 0.1@u_Ω)  # 导线电阻很小

sim2 = circuit_short.simulator(temperature=25, nominal_temperature=25,
                               simulator='ngspice-subprocess')
analysis2 = sim2.operating_point()

v2 = float(analysis2.nodes['1'][0])
i2 = v2 / 0.1
p2 = v2 * i2

print()
print("【短路电路】导线直接连接 (220V, 0.1Ω)")
print("  电流 I = {:.0f} A !!".format(i2))
print("  功率 P = {:.0f} W !!".format(p2))
print("  -> 这是极其危险的！会导致导线熔化和火灾")
print()
print("【结论】")
print("  焦耳定律 Q = I²Rt 告诉我们：")
print("  - 电流越大，发热越严重（平方关系！）")
print("  - 短路时电阻极小，电流极大，发热剧烈")
print("  - 这就是为什么电路中必须有保险丝或断路器")
